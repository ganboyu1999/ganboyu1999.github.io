cmake_minimum_required(VERSION 3.20)
project(first_app LANGUAGES CXX)
find_package(frs CONFIG REQUIRED COMPONENTS runtime)
find_library(FRS_CLIENT_LIBRARY NAMES frs_sdk_cpp HINTS "${frs_LIBRARY_DIR}" REQUIRED)
find_library(FRS_SERVICE_LIBRARY NAMES frs_core_service HINTS "${frs_LIBRARY_DIR}" REQUIRED)
add_executable(first_app main.cpp)
target_compile_features(first_app PRIVATE cxx_std_17)
target_compile_options(first_app PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(first_app PRIVATE "${FRS_CLIENT_LIBRARY}" "${FRS_SERVICE_LIBRARY}" frs::runtime)
