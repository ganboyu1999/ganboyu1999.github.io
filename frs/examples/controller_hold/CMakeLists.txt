cmake_minimum_required(VERSION 3.20)
project(tutorial_hold LANGUAGES CXX)
find_package(frs CONFIG REQUIRED COMPONENTS atom)
add_library(tutorial_hold SHARED main.cpp)
target_compile_features(tutorial_hold PRIVATE cxx_std_17)
target_compile_options(tutorial_hold PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(tutorial_hold PRIVATE frs::controller_module)
set_target_properties(tutorial_hold PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
configure_file(controller.json controller.json COPYONLY)
