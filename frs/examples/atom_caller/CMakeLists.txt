cmake_minimum_required(VERSION 3.20)
project(atom_caller LANGUAGES CXX)
find_package(frs CONFIG REQUIRED COMPONENTS runtime)
find_library(FRS_CLIENT_LIBRARY NAMES frs_sdk_cpp HINTS "${frs_LIBRARY_DIR}" REQUIRED)
find_library(FRS_SERVICE_LIBRARY NAMES frs_core_service HINTS "${frs_LIBRARY_DIR}" REQUIRED)
add_executable(atom_caller main.cpp)
target_compile_features(atom_caller PRIVATE cxx_std_17)
target_compile_options(atom_caller PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(atom_caller PRIVATE "${FRS_CLIENT_LIBRARY}" "${FRS_SERVICE_LIBRARY}" frs::runtime)
